// 시나리오 작업실 → 구글 문서/스프레드시트 자동 저장
// 앱의 "구글 연결" 창에서 "코드 복사"를 누르면 KEY가 채워진 이 코드가 복사돼요.
const KEY = '__KEY__';
const FOLDER = '시나리오 작업실';
const VERSION = 4;

function doGet() { return out({ ok: true, app: 'screenwriter-sync', version: VERSION }); }

let STEP = '시작';
function doPost(e) {
  try {
    const d = JSON.parse(e.postData.contents);
    if (KEY && d.key !== KEY) return out({ ok: false, error: '비밀 키가 맞지 않아요' });
    if (d.ping) return out({ ok: true, version: VERSION });
    const folder = projectFolder(d.title);
    const props = PropertiesService.getScriptProperties();
    const ids = JSON.parse(props.getProperty('p:' + d.id) || '{}');

    const save = function () { props.setProperty('p:' + d.id, JSON.stringify(ids)); };
    const hash = function (x) { return Utilities.base64Encode(Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, JSON.stringify(x), Utilities.Charset.UTF_8)); };

    // 바뀐 것만 다시 씀 (시간 아끼기). 한 단계씩 저장해서 중간에 멈춰도 다음에 이어서 함
    STEP = '대본 문서';
    const scriptDoc = openOrCreateDoc(ids.script, d.title + ' - 대본', folder);
    if (ids.script !== scriptDoc.getId()) { ids.script = scriptDoc.getId(); ids.hScript = ''; }
    scriptDoc.setName(d.title + ' - 대본');
    const scriptUrl = scriptDoc.getUrl();
    const hs = hash([d.title, d.script]);
    if (ids.hScript !== hs) { writeScript(scriptDoc, d.title, d.script); ids.hScript = hs; }
    save();

    STEP = '장면 묘사 문서';
    const descDoc = openOrCreateDoc(ids.desc, d.title + ' - 장면 묘사', folder);
    if (ids.desc !== descDoc.getId()) { ids.desc = descDoc.getId(); ids.hDesc = ''; }
    descDoc.setName(d.title + ' - 장면 묘사');
    const descUrl = descDoc.getUrl();
    const hd = hash([d.title, d.describeText]);
    if (ids.hDesc !== hd) { writeLines(descDoc, d.title + ' 장면 묘사', d.describeText); ids.hDesc = hd; }
    save();

    STEP = '스프레드시트';
    const ss = openOrCreateSheet(ids.sheet, d.title + ' - 신 리스트', folder);
    if (ids.sheet !== ss.getId()) { ids.sheet = ss.getId(); ids.hSheet = ''; }
    ss.rename(d.title + ' - 신 리스트');
    const hh = hash([VERSION, d.tree]);
    if (ids.hSheet !== hh) {
      STEP = '신 리스트 탭';
      writeSceneList(ss, d.tree);
      STEP = '장면 묘사 탭';
      writeDescribe(ss, d.tree);
      STEP = '마무리';
      ['컷 리스트', 'Sheet1', '시트1'].forEach(function (n) { const x = ss.getSheetByName(n); if (x && ss.getSheets().length > 1) ss.deleteSheet(x); });
      ss.setActiveSheet(ss.getSheetByName('신 리스트')); ss.moveActiveSheet(1);
      SpreadsheetApp.flush();
      ids.hSheet = hh;
    }
    save();
    return out({ ok: true, version: VERSION, script: scriptUrl, desc: descUrl, sheet: ss.getUrl() });
  } catch (err) {
    return out({ ok: false, error: '[' + STEP + '] ' + String(err) });
  }
}

// 문서를 비운 뒤 첫 문단 (없으면 새로 만듦)
function firstPara(body) {
  const ps = body.getParagraphs();
  return ps.length ? ps[0] : body.appendParagraph('');
}

function out(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }

function projectFolder(title) {
  const roots = DriveApp.getFoldersByName(FOLDER);
  const root = roots.hasNext() ? roots.next() : DriveApp.createFolder(FOLDER);
  return root;
}
function alive(id) { try { const f = DriveApp.getFileById(id); return !f.isTrashed(); } catch (e) { return false; } }
function openOrCreateDoc(id, name, folder) {
  if (id && alive(id)) return DocumentApp.openById(id);
  const doc = DocumentApp.create(name);
  DriveApp.getFileById(doc.getId()).moveTo(folder);
  return doc;
}
function openOrCreateSheet(id, name, folder) {
  if (id && alive(id)) return SpreadsheetApp.openById(id);
  const ss = SpreadsheetApp.create(name);
  DriveApp.getFileById(ss.getId()).moveTo(folder);
  return ss;
}

function writeScript(doc, title, paras) {
  const body = doc.getBody();
  body.clear();
  body.setMarginLeft(56); body.setMarginRight(56);
  const h = firstPara(body);
  h.setText(title);
  h.setHeading(DocumentApp.ParagraphHeading.TITLE);
  h.setAlignment(DocumentApp.HorizontalAlignment.CENTER);
  const A = DocumentApp.HorizontalAlignment;
  paras.forEach(function (x) {
    const p = body.appendParagraph(x.text || ' ');
    p.setHeading(DocumentApp.ParagraphHeading.NORMAL);
    p.setIndentStart(0); p.setIndentFirstLine(0); p.setIndentEnd(0);
    p.setAlignment(A.LEFT); p.setSpacingBefore(0); p.setSpacingAfter(4);
    const t = p.editAsText();
    t.setBold(false); t.setItalic(false); t.setFontSize(11);
    if (x.t === 'scene') { t.setBold(true); t.setFontSize(12); p.setSpacingBefore(18); }
    else if (x.t === 'character') { t.setBold(true); p.setIndentStart(150); p.setIndentFirstLine(150); p.setSpacingBefore(8); p.setSpacingAfter(0); }
    else if (x.t === 'dialogue') { p.setIndentStart(80); p.setIndentFirstLine(80); p.setIndentEnd(80); }
    else if (x.t === 'paren') { t.setItalic(true); p.setIndentStart(115); p.setIndentFirstLine(115); p.setSpacingAfter(0); }
    else if (x.t === 'transition') { p.setAlignment(A.RIGHT); }
    else if (x.t === 'shot') { t.setBold(true); }
  });
  doc.saveAndClose();
}
function writeLines(doc, title, lines) {
  const body = doc.getBody();
  body.clear();
  const h = firstPara(body);
  h.setText(title);
  h.setHeading(DocumentApp.ParagraphHeading.TITLE);
  lines.forEach(function (l) {
    const p = body.appendParagraph(l || ' ');
    p.setHeading(/^S#\d+/.test(l) ? DocumentApp.ParagraphHeading.HEADING2 : DocumentApp.ParagraphHeading.NORMAL);
  });
  doc.saveAndClose();
}
// 시트를 새로 만들어 바꿔 끼움 (예전 묶음·서식이 남지 않게)
function freshSheet(ss, name) {
  const old = ss.getSheetByName(name);
  const left = ss.getSheetByName(name + ' (새로)'); // 지난번에 중간에 멈췄을 때 남은 것
  if (left) ss.deleteSheet(left);
  const tmp = ss.insertSheet(name + ' (새로)');
  if (old) ss.deleteSheet(old);
  tmp.setName(name);
  return tmp;
}

// 공통: 표 그리기. rows = [{ v: [...], kind: 'head'|'scene'|'cut' }]
function drawTable(sh, widths, wrapCols, rows, groups) {
  const w = widths.length;
  const vals = rows.map(function (r) { const a = r.v.map(function (x) { return x === null || x === undefined ? '' : x; }); while (a.length < w) a.push(''); return a; });
  if (sh.getMaxColumns() < w) sh.insertColumnsAfter(sh.getMaxColumns(), w - sh.getMaxColumns());
  if (sh.getMaxRows() < vals.length + 1) sh.insertRowsAfter(sh.getMaxRows(), vals.length + 1 - sh.getMaxRows());
  const all = sh.getRange(1, 1, vals.length, w);
  all.setNumberFormat('@');
  all.setValues(vals);
  all.setFontSize(10); all.setVerticalAlignment('top');
  all.setWrapStrategy(SpreadsheetApp.WrapStrategy.CLIP);
  const bg = [], fw = [], fc = [];
  rows.forEach(function (r) {
    const b = r.kind === 'head' ? '#222222' : r.kind === 'scene' ? '#f4f4f0' : '#ffffff';
    const c = r.kind === 'head' ? '#ffffff' : r.kind === 'cut' ? '#333333' : '#111111';
    const f = r.kind === 'cut' ? 'normal' : 'bold';
    const rb = [], rf = [], rc = []; for (let i = 0; i < w; i++) { rb.push(b); rf.push(f); rc.push(c); }
    bg.push(rb); fw.push(rf); fc.push(rc);
  });
  all.setBackgrounds(bg); all.setFontWeights(fw); all.setFontColors(fc);
  all.setBorder(true, true, true, true, false, true, '#dddddd', SpreadsheetApp.BorderStyle.SOLID);
  widths.forEach(function (px, i) { sh.setColumnWidth(i + 1, px); });
  wrapCols.forEach(function (c) { if (vals.length > 1) sh.getRange(2, c, vals.length - 1, 1).setWrapStrategy(SpreadsheetApp.WrapStrategy.WRAP); });
  sh.setFrozenRows(1); sh.setFrozenColumns(1);
  sh.setRowHeight(1, 30);
  // 씬 아래 컷 줄 묶기 (앱의 ▸ 펼치기처럼, 처음엔 접힘)
  sh.setRowGroupControlPosition(SpreadsheetApp.GroupControlTogglePosition.BEFORE);
  groups.forEach(function (g) { if (g.n > 0) sh.getRange(g.start, 1, g.n, 1).shiftRowGroupDepth(1); });
  groups.forEach(function (g) { if (g.n > 0) { try { sh.getRowGroup(g.start, 1).collapse(); } catch (e) {} } });
  const extraCols = sh.getMaxColumns() - w; if (extraCols > 0) sh.deleteColumns(w + 1, extraCols);
  const extraRows = sh.getMaxRows() - vals.length; if (extraRows > 1) sh.deleteRows(vals.length + 2, extraRows - 1);
}

// 신 리스트: 씬 한 줄 + (접힌) 컷 줄
function writeSceneList(ss, tree) {
  const sh = freshSheet(ss, '신 리스트');
  const rows = [{ kind: 'head', v: ['S# / 컷', '장소', '시간', '등장인물', '내용', '컷', '분량', '대사', '비고', '샷'] }];
  const groups = [];
  (tree || []).forEach(function (s) {
    rows.push({ kind: 'scene', v: ['S#' + s.num, s.place, s.time, s.chars, s.summary, s.cutCount, s.pages + '쪽 · ' + s.secs + '초', s.dlg, s.notes, ''] });
    const start = rows.length + 1;
    s.cuts.forEach(function (c) {
      const body = [c.desc || c.text, c.dialogue].filter(function (x) { return x; }).join('\n');
      rows.push({ kind: 'cut', v: ['   ' + c.label, '', '', c.chars, body, '', c.secs + '초', '', '', c.shot] });
    });
    groups.push({ start: start, n: s.cuts.length });
  });
  drawTable(sh, [70, 170, 60, 150, 460, 45, 90, 45, 180, 170], [5, 9], rows, groups);
}

// 장면 묘사: 씬(설정표) + 컷별 [대본 | 장면 묘사 | 바로잡기]
function writeDescribe(ss, tree) {
  const sh = freshSheet(ss, '장면 묘사');
  const rows = [{ kind: 'head', v: ['S# / 컷', '대본', '장면 묘사', '바로잡기', '샷'] }];
  const groups = [];
  (tree || []).forEach(function (s) {
    rows.push({ kind: 'scene', v: ['S#' + s.num, s.place + (s.time ? ' / ' + s.time : ''), s.setTable ? '[설정표] ' + s.setTable : '', '', ''] });
    const start = rows.length + 1;
    s.cuts.forEach(function (c) {
      rows.push({ kind: 'cut', v: ['   ' + c.label, [c.text, c.dialogue].filter(function (x) { return x; }).join('\n'), c.desc, c.corr, c.shot] });
    });
    groups.push({ start: start, n: s.cuts.length });
  });
  drawTable(sh, [70, 380, 460, 200, 160], [2, 3, 4], rows, groups);
}
